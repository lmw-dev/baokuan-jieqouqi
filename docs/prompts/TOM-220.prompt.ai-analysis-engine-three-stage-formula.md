### 3. Prompt管理器

```python
import os
import aiofiles
from pathlib import Path
from typing import Dict

class PromptManager:
    """Prompt模板管理器"""
    
    def __init__(self, templates_dir: str):
        self.templates_dir = Path(templates_dir)
        self.templates: Dict[str, PromptTemplate] = {}
    
    async def load_template(self, name: str) -> PromptTemplate:
        """从MD文件加载Prompt模板"""
        file_path = self.templates_dir / f"{name}.md"
        if not file_path.exists():
            raise FileNotFoundError(f"Template {name} not found")
        
        async with aiofiles.open(file_path, 'r', encoding='utf-8') as f:
            content = await f.read()
        
        # 解析MD文档，提取版本和内容
        return self._parse_template(name, content)
    
    async def reload_templates(self, names: Optional[List[str]] = None):
        """重新加载指定模板"""
        if names is None:
            names = [f.stem for f in self.templates_dir.glob("*.md")]
        
        for name in names:
            template = await self.load_template(name)
            self.templates[name] = template
    
    def get_template(self, name: str, version: str = "latest") -> str:
        """获取Prompt模板内容"""
        if name not in self.templates:
            raise ValueError(f"Template {name} not loaded")
        return self.templates[name].content
```

### 4. 分析服务

```python
class ThreeStageAnalyzer:
    """三段式分析器"""
    
    def __init__(self, ai_service: AIServiceBase, prompt_manager: PromptManager):
        self.ai_service = ai_service
        self.prompt_manager = prompt_manager
    
    async def analyze(self, request: AnalysisRequest) -> ThreeStageAnalysisResult:
        """执行三段式分析"""
        start_time = time.time()
        
        # 并行执行三个分析模块
        trust_hook_task = self._analyze_trust_hook(request.content)
        empathy_anchor_task = self._analyze_empathy_anchor(request.content)
        value_map_task = self._analyze_value_map(request.content)
        
        trust_hook, empathy_anchor, value_map = await asyncio.gather(
            trust_hook_task, empathy_anchor_task, value_map_task
        )
        
        processing_time = int((time.time() - start_time) * 1000)
        
        return ThreeStageAnalysisResult(
            trust_hook=trust_hook,
            empathy_anchor=empathy_anchor,
            value_map=value_map,
            meta=AnalysisMeta(
                analysis_id=str(uuid.uuid4()),
                timestamp=datetime.utcnow(),
                processing_time_ms=processing_time,
                ai_model=self.ai_service.model,
                prompt_version=request.prompt_version or "latest"
            )
        )
```

- ✅ **正确做法**: 使用不可变对象或适当的同步机制

#### 4. 资源泄漏预防
- ❌ **错误做法**: HTTP连接不正确关闭
- ✅ **正确做法**: 使用async with或try-finally确保资源释放
